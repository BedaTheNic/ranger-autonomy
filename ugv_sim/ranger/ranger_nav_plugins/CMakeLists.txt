# Humble MPPI exposes xtensor types without exporting its ABI compile flags.
# Match the installed xsimd allocator/alignment (32 bytes on amd64 AVX2).
function(ranger_mppi_abi target)
  target_compile_definitions(${target} PRIVATE XTENSOR_ENABLE_XSIMD XTENSOR_USE_XSIMD)
  target_compile_options(${target} PRIVATE -O2)
  if(CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|AMD64|amd64")
    target_compile_options(${target} PRIVATE -mno-avx512f -msse4.2 -mavx2 -mfma)
  endif()
  target_link_libraries(${target} xtensor xtensor::use_xsimd)
endfunction()
